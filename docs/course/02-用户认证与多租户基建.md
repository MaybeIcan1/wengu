## 🗄️ 数据库：t_user（你亲手在 Navicat 中执行）

新建查询 → 粘贴执行到 `wengu` 库：

```sql
CREATE TABLE t_user (
  id            BIGINT       NOT NULL COMMENT '雪花ID',
  username      VARCHAR(50)  NOT NULL COMMENT '登录名',
  password      VARCHAR(100) NOT NULL COMMENT 'BCrypt密文',
  nickname      VARCHAR(50)  DEFAULT NULL,
  email         VARCHAR(100) DEFAULT NULL,
  created_time  DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_time  DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  deleted       TINYINT      NOT NULL DEFAULT 0 COMMENT '逻辑删除 0正常 1删除',
  PRIMARY KEY (id),
  UNIQUE KEY uk_username (username)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='用户表（全局表，无user_id列）';
```

**设计理由（面试问答就出自这里）**：
- `id` 用雪花而非自增：为以后分库分表预留全局唯一能力；自增 ID 在迁移/合并库时会撞号
- `password` 存 BCrypt **密文**：慢哈希+内置盐，数据库拖库也无法反推明文（MD5+彩虹表神仙难救）
- `uk_username` 唯一索引：让数据库替我们挡并发重复注册（代码层校验只是第一道）
- `created_time/updated_time` 数据库给默认值兜底，应用层还配自动填充（双保险）
- ★ **本表没有 user_id 列**：它自己就是"租户"，后面所有业务表第一列都是 user_id——这是多租户数据隔离的物理基础
- `deleted` 逻辑删除：删用户=标记，审计/恢复友好；8.4 中 `TINYINT` 只占 1 字节

```java
package com.wengu.backend.common.result;

import lombok.Data;

@Data
public class Result<T> {
    private Integer code;   // 200=成功 400=参数/业务错 401=未登录 429=限流 500=系统错
    private String message;
    private T data;

    public static <T> Result<T> success()              { return build(200, "ok", null); }
    public static <T> Result<T> success(T data)        { return build(200, "ok", data); }
    public static <T> Result<T> fail(int code, String msg) { return build(code, msg, null); }
    private static <T> Result<T> build(int code, String msg, T data) {
        Result<T> r = new Result<>();
        r.code = code; r.message = msg; r.data = data;
        return r;
    }
}
```

```java
package com.wengu.backend.common.exception;

import lombok.extern.slf4j.Slf4j;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import com.wengu.backend.common.result.Result;
import java.util.stream.Collectors;

@Slf4j
@RestControllerAdvice
public class GlobalExceptionHandler {

    @ExceptionHandler(BizException.class)
    public Result<Void> biz(BizException e) {
        return Result.fail(e.getCode(), e.getMessage());
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public Result<Void> valid(MethodArgumentNotValidException e) {
        String msg = e.getBindingResult().getFieldErrors().stream()
                .map(f -> f.getField() + ": " + f.getDefaultMessage())
                .collect(Collectors.joining("; "));
        return Result.fail(400, msg);
    }

    @ExceptionHandler(Exception.class)
    public Result<Void> other(Exception e) {
        log.error("unhandled", e);                    // 全栈进日志
        return Result.fail(500, "系统繁忙，请稍后重试"); // 对外只说人话
    }
}
```

```java
// common/util/JwtUtil.java
package com.wengu.backend.common.util;

import io.jsonwebtoken.Jwts;
import javax.crypto.SecretKey;
import io.jsonwebtoken.security.Keys;
import java.nio.charset.StandardCharsets;
import java.time.Duration;
import java.util.Date;

/** ★ 秘钥先常量，Phase 8 上云时改为环境变量注入（讲评时问你怎么改） */
public class JwtUtil {
    private static final SecretKey KEY = Keys.hmacShaKeyFor(
            "wengu-dev-secret-key-please-32-bytes!!".getBytes(StandardCharsets.UTF_8));
    private static final long TTL = Duration.ofDays(7).toMillis();

    public static String gen(Long userId) {
        return Jwts.builder()
                .subject(String.valueOf(userId))
                .expiration(new Date(System.currentTimeMillis() + TTL))
                .signWith(KEY)
                .compact();
    }
    /** 失败返回 null（抛出去会让拦截器写重复的401逻辑） */
    public static Long parse(String token) {
        try {
            return Long.valueOf(Jwts.parser().verifyWith(KEY).build()
                    .parseSignedClaims(token).getPayload().getSubject());
        } catch (Exception e) {
            return null;
        }
    }
}
```

```java
import com.baomidou.mybatisplus.annotation.DbType;
import com.baomidou.mybatisplus.core.handlers.MetaObjectHandler;
import com.baomidou.mybatisplus.extension.plugins.MybatisPlusInterceptor;

import com.baomidou.mybatisplus.extension.plugins.inner.PaginationInnerInterceptor;
import com.baomidou.mybatisplus.extension.plugins.inner.TenantLineInnerInterceptor;
import com.wengu.backend.common.util.UserContext;
import net.sf.jsqlparser.expression.Expression;
import net.sf.jsqlparser.expression.LongValue;
import org.apache.ibatis.reflection.MetaObject;
import org.mybatis.spring.annotation.MapperScan;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import java.time.LocalDateTime;
import java.util.Set;

@Configuration
@MapperScan("com.wengu.backend.**.mapper")   // 扫描所有模块下的mapper包
public class MybatisPlusConfig {

    @Bean
    public MybatisPlusInterceptor mybatisPlusInterceptor() {
        MybatisPlusInterceptor interceptor = new MybatisPlusInterceptor();
        // ① 多租户插件：所有SQL自动拼 WHERE user_id = ?（数据隔离的大杀器）
        interceptor.addInnerInterceptor(new TenantLineInnerInterceptor(new TenantLineHandler() {
            @Override
            public Expression getTenantId() {
                return new LongValue(UserContext.requireUserId());
            }
            @Override
            public String getTenantIdColumn() { return "user_id"; }
            @Override
            public boolean ignoreTable(String tableName) {
                // 没有user_id列的表要跳过；★ t_user 是"租户本体"故在列
                return Set.of("t_user").contains(tableName);
            }
        }));
        // ② 分页插件（第3章的列表接口用）
        interceptor.addInnerInterceptor(new PaginationInnerInterceptor(DbType.MYSQL));
        return interceptor;
    }

    /** created_time/updated_time 自动填充 */
    @Bean
    public MetaObjectHandler metaObjectHandler() {
        return new MetaObjectHandler() {
            @Override
            public void insertFill(MetaObject metaObject) {
                strictInsertFill(metaObject, "createdTime", LocalDateTime.class, LocalDateTime.now());
                strictInsertFill(metaObject, "updatedTime", LocalDateTime.class, LocalDateTime.now());
            }
            @Override
            public void updateFill(MetaObject metaObject) {
                strictUpdateFill(metaObject, "updatedTime", LocalDateTime.class, LocalDateTime.now());
            }
        };
    }
}
```

```java
// security/LoginInterceptor.java
package com.wengu.backend.security;

import com.wengu.backend.common.util.JwtUtil;
import com.wengu.backend.common.util.UserContext;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.web.servlet.HandlerInterceptor;

import java.nio.charset.StandardCharsets;

public class LoginInterceptor implements HandlerInterceptor {

    @Override
    public boolean preHandle(HttpServletRequest req, HttpServletResponse resp, Object handler)
            throws Exception {
        String auth = req.getHeader("Authorization");
        if (auth == null || !auth.startsWith("Bearer ")) return deny(resp);
        Long userId = JwtUtil.parse(auth.substring(7));
        if (userId == null) return deny(resp);
        UserContext.set(userId);          // ★ 本请求线程后面的代码都"知道"你是谁
        return true;
    }

    @Override
    public void afterCompletion(HttpServletRequest req, HttpServletResponse resp,
                                Object handler, Exception ex) {
        UserContext.clear();              // ★ Tomcat线程池会复用线程，不清=串号，内存泄漏
    }

    private boolean deny(HttpServletResponse resp) throws java.io.IOException {
        resp.setStatus(401);
        resp.setContentType("application/json;charset=UTF-8");
        resp.getOutputStream().write(
                "{\"code\":401,\"message\":\"未登录或登录已过期\",\"data\":null}"
                        .getBytes(StandardCharsets.UTF_8));
        return false;
    }
}
```

```java
// security/service/UserServiceImpl.java（核心业务全在这，先读懂再抄）
package com.wengu.backend.security.service;

import com.baomidou.mybatisplus.spring.service.impl.ServiceImpl;
import com.wengu.backend.security.dto.*;
import com.wengu.backend.security.entity.User;
import com.wengu.backend.security.mapper.UserMapper;
import com.wengu.backend.security.vo.*;
import com.wengu.backend.common.exception.BizException;
import com.wengu.backend.common.util.JwtUtil;
import lombok.RequiredArgsConstructor;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.stereotype.Service;

@Service
@RequiredArgsConstructor
public class UserServiceImpl extends ServiceImpl<UserMapper, User> implements UserService {

    private final BCryptPasswordEncoder encoder = new BCryptPasswordEncoder();

    @Override
    public void register(RegisterDTO dto) {
        // 1) 业务校验（撞库读密码法：登录时统一报"用户名或密码错误"）
        Long count = lambdaQuery().eq(User::getUsername, dto.getUsername()).count();
        if (count > 0) throw new BizException(400, "用户名已被占用");
        // 2) 组装 + 加密 + 落库
        User u = new User();
        u.setUsername(dto.getUsername());
        u.setPassword(encoder.encode(dto.getPassword()));   // BCrypt: 明文进不了库
        u.setNickname(dto.getNickname() == null ? dto.getUsername() : dto.getNickname());
        save(u);
    }

    @Override
    public LoginVO login(LoginDTO dto) {
        User user = lambdaQuery().eq(User::getUsername, dto.getUsername()).one();
        if (user == null || !encoder.matches(dto.getPassword(), user.getPassword()))
            throw new BizException(400, "用户名或密码错误");
        return new LoginVO(JwtUtil.gen(user.getId()), user.getId(), user.getNickname());
    }

    @Override
    public UserVO me(Long userId) {
        User u = getById(userId);
        if (u == null) throw new BizException(401, "用户不存在");
        return UserVO.builder()
                .id(u.getId()).username(u.getUsername())
                .nickname(u.getNickname()).email(u.getEmail())
                .build();                                  // ★ password 永远不进 VO
    }
}
```

```ts
import axios from 'axios'
import { ElMessage } from 'element-plus'
import { useUserStore } from '../stores/user'
import router from '../router'

const request = axios.create({ baseURL: '/api', timeout: 15000 })

request.interceptors.request.use(cfg => {
  const token = useUserStore().token
  if (token) cfg.headers.Authorization = `Bearer ${token}`
  return cfg
})
request.interceptors.response.use(resp => {
  const { code, message, data } = resp.data
  if (code === 200) return data
  if (code === 401) { router.push('/login'); return Promise.reject() }
  ElMessage.error(message || '请求失败')
  return Promise.reject(new Error(message))
}, err => { ElMessage.error('网络异常'); return Promise.reject(err) })

export default request
```

```ts
// stores/user.ts
import { defineStore } from 'pinia'
import { loginApi } from '../api/user'

export const useUserStore = defineStore('user', {
  state: () => ({
    token: localStorage.getItem('wengu_token') || '',
    userId: 0, nickname: '' as string,
  }),
  actions: {
    async login(payload: { username: string; password: string }) {
      const vo: any = await loginApi(payload)          // axios层已拆包，拿到的就是data
      this.token = vo.token; this.userId = vo.userId; this.nickname = vo.nickname
      localStorage.setItem('wengu_token', vo.token)
    },
    logout() {
      this.token = ''; localStorage.removeItem('wengu_token'); location.href = '/login'
    },
  },
})
```

> 💣 三个前端实战坑（2026-10-17 实录）：
> 1. **箭头函数的 `=>` 不能在行首**（TS1200）：正确写法是 `const f = (...) =>\n  request.post(...)`，`=>` 留在行尾
> 2. **模板项目的 tsconfig 开了 noUnusedLocals**：import 了没用上的东西直接编译报错——这不是坏事，帮你保持 import 干净

## 🔗 关联八股

- BCrypt 与 MD5：慢哈希+随机盐，抗彩虹表；「登录失败信息统一」防账号枚举
- JWT 三段 header.payload.signature：签名防篡改≠加密；与 Session/Cookie 差异（你笔记 `①Java中==` 旁边那些都能挂上）
- ThreadLocal：本线程变量；线程池复用 → 必须 afterCompletion clear（8 章还会在压测里见到它变成泄漏点）
- 多租户插件：MyBatis-Plus 拦截器=责任链改写 SQL（八股：插件执行原理）；好处=一行都不用手写 user_id，坏处=忘配 ignoreTable 时 t_user 查询会报错
- 构造器注入 vs 字段注入：final 不可变 + 便于单测；@Autowired 字段注入 hide dependencies
- 事务注解 `rollbackFor=Exception.class`：默认只回滚 RuntimeException；第 3 章首次使用
- 逻辑删除 vs 物理删除：审计可恢复 vs 索引残留（deleted 也进唯一索引的设计坑以后讲）
